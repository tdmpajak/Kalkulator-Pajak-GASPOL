/** * GASPOL — Tanya Pajak Chatbot: Backend Apps Script * --------------------------------------------------- * Fungsi: * - Menerima pertanyaan yang tidak terjawab dari chatbot (doPost) → sheet "Pertanyaan_CS" * - Nomor HP otomatis diformat jadi link WhatsApp, Email jadi link mailto (klik langsung dari Sheets) * - Mengirim EMAIL NOTIFIKASI otomatis ke Tim Tax setiap ada pertanyaan baru masuk * - Begitu kolom "Jawaban" di sheet "Pertanyaan_CS" diisi, JAWABANNYA OTOMATIS DIKIRIM EMAIL * ke penanya (kalau dia mengisi Email waktu bertanya) — cukup ketik jawaban di Sheets, selesai. * - Mencatat statistik penggunaan portal GASPOL (menu apa yang paling sering dipakai) → sheet "Log_Penggunaan" * * CARA PASANG: * 1. Buka Google Sheets yang akan dipakai sebagai database * (disarankan pakai spreadsheet yang sama dengan NC Tracker, akun tdm.pajak10@gmail.com) * 2. Extensions > Apps Script, hapus isi default, tempel seluruh isi file ini. * 3. Isi ADMIN_EMAIL di bawah ini dengan email Anda (untuk terima notifikasi pertanyaan baru). * 4. Jalankan fungsi `setupSheet` sekali (pilih dari dropdown function lalu klik Run) — * ini otomatis membuat sheet "Pertanyaan_CS" & "Log_Penggunaan" BESERTA trigger auto-email jawaban. * Saat pertama kali Run, akan muncul pop-up minta izin — klik Allow/Izinkan semuanya * (termasuk izin kirim email, kalau tidak di-Allow maka notifikasi TIDAK akan jalan). * 5. Deploy > New deployment > Type: Web app. * - Execute as: Me * - Who has access: Anyone * 6. Salin URL Web App yang muncul (formatnya .../exec) * 7. Tempel URL tersebut ke variabel CS_SCRIPT_URL di index.html (bagian atas modul chatbot) * * KALAU EDIT KODE INI DI KEMUDIAN HARI: * Setelah edit & save, deployment yang sudah aktif TIDAK otomatis ter-update. Harus: * Deploy > Manage deployments > klik ikon pensil pada deployment yang ada > Version: "New version" > Deploy. * ("New deployment" akan bikin URL baru — hindari kalau URL lama sudah dipakai di index.html.) */ // ⚠️ WAJIB DIISI — email yang akan menerima notifikasi setiap ada pertanyaan baru dari chatbot const ADMIN_EMAIL = 'isi-email-anda@gmail.com'; const SHEET_NAME = 'Pertanyaan_CS'; const HEADERS = ['Timestamp', 'Nama', 'No HP', 'WA Link', 'Email', 'Email Link', 'Pertanyaan', 'Sumber', 'Status', 'Jawaban', 'Dijawab Oleh', 'Waktu Jawab']; const LOG_SHEET_NAME = 'Log_Penggunaan'; const LOG_HEADERS = ['Timestamp', 'Halaman/Fitur', 'Detail']; const STATUS_SHEET_NAME = 'Status_CS'; // index kolom (1-based) di sheet Pertanyaan_CS, sesuai urutan HEADERS const COL_STATUS = 9; const COL_JAWABAN = 10; const COL_DIJAWAB_OLEH = 11; const COL_WAKTU_JAWAB = 12; const COL_EMAIL = 5; const COL_NAMA = 2; const COL_PERTANYAAN = 7; function setupSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) sheet = ss.insertSheet(SHEET_NAME); sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold').setBackground('#C81E2C').setFontColor('#FFFFFF'); sheet.setFrozenRows(1); sheet.autoResizeColumns(1, HEADERS.length); let logSheet = ss.getSheetByName(LOG_SHEET_NAME); if (!logSheet) logSheet = ss.insertSheet(LOG_SHEET_NAME); logSheet.getRange(1, 1, 1, LOG_HEADERS.length).setValues([LOG_HEADERS]); logSheet.getRange(1, 1, 1, LOG_HEADERS.length).setFontWeight('bold').setBackground('#C81E2C').setFontColor('#FFFFFF'); logSheet.setFrozenRows(1); logSheet.autoResizeColumns(1, LOG_HEADERS.length); setupRingkasanUtilisasi(ss); setupStatusSheet(ss); setupEditTrigger(); } /** Bikin sheet "Status_CS" — Tim Tax centang/hilangkan centang di sini untuk menandakan * sedang standby (online) balas chat atau tidak. Dibaca oleh chatbot lewat action "check_status". */ function setupStatusSheet(ss) { let sh = ss.getSheetByName(STATUS_SHEET_NAME); if (!sh) sh = ss.insertSheet(STATUS_SHEET_NAME); sh.getRange('A1').setValue('CS Online? (centang kalau sedang standby balas chat GASPOL secara langsung)'); sh.getRange('A1').setFontWeight('bold').setWrap(true); const cell = sh.getRange('B1'); if (cell.getValue() === '') cell.setValue(false); const rule = SpreadsheetApp.newDataValidation().requireCheckbox().build(); cell.setDataValidation(rule); sh.setColumnWidth(1, 420); sh.setColumnWidth(2, 70); sh.getRange('A3').setValue('Centang B1 kapan pun Anda sedang siap membalas pertanyaan chatbot secara langsung.\nKalau tidak dicentang, pertanyaan tetap masuk seperti biasa dan dijawab dalam 1×24 jam.'); sh.getRange('A3').setWrap(true).setFontColor('#888888').setFontSize(9); } /** Bikin (atau perbarui) trigger otomatis: begitu kolom "Jawaban" di sheet Pertanyaan_CS * diisi manual oleh Tim Tax, jawabannya langsung dikirim email ke penanya. Simple trigger * bawaan (onEdit) TIDAK boleh kirim email karena keterbatasan izin, makanya harus pakai * installable trigger seperti ini — cukup dijalankan sekali lewat setupSheet(). */ function setupEditTrigger() { const triggers = ScriptApp.getProjectTriggers(); triggers.forEach(t => { if (t.getHandlerFunction() === 'onEditInstallable') ScriptApp.deleteTrigger(t); }); const ss = SpreadsheetApp.getActiveSpreadsheet(); ScriptApp.newTrigger('onEditInstallable') .forSpreadsheet(ss) .onEdit() .create(); } /** Installable trigger — jalan otomatis tiap ada perubahan di spreadsheet manapun. * Kalau yang diedit adalah kolom "Jawaban" di sheet Pertanyaan_CS dan baris itu punya * Email, kirim email jawabannya ke penanya + update Status & Waktu Jawab otomatis. */ function onEditInstallable(e) { try { if (!e || !e.range) return; const sheet = e.range.getSheet(); if (sheet.getName() !== SHEET_NAME) return; if (e.range.getColumn() !== COL_JAWABAN) return; const row = e.range.getRow(); if (row === 1) return; // header const jawaban = e.range.getValue().toString().trim(); if (!jawaban) return; // kolom dikosongkan lagi, jangan kirim apa-apa const rowVals = sheet.getRange(row, 1, 1, HEADERS.length).getValues()[0]; const nama = rowVals[COL_NAMA - 1]; const email = rowVals[COL_EMAIL - 1]; const pertanyaan = rowVals[COL_PERTANYAAN - 1]; // update Status & Waktu Jawab otomatis, terlepas dari ada email atau tidak sheet.getRange(row, COL_STATUS).setValue('Selesai'); sheet.getRange(row, COL_WAKTU_JAWAB).setValue(new Date()); try { sheet.getRange(row, COL_DIJAWAB_OLEH).setValue(Session.getActiveUser().getEmail() || ''); } catch (ignoreErr) { /* kadang tidak bisa diambil, biarkan kosong */ } if (!email) return; // penanya tidak isi email — cukup dibalas manual via WA (link sudah ada di kolom WA Link) const subject = 'Jawaban pertanyaan Anda di GASPOL'; const body = 'Halo ' + nama + ',\n\n' + 'Terkait pertanyaan Anda di GASPOL:\n"' + pertanyaan + '"\n\n' + 'Jawaban dari Tim Tax TDM:\n' + jawaban + '\n\n' + 'Terima kasih,\nTim Tax & Artefac — PT Tunas Dwipa Matra'; MailApp.sendEmail(email, subject, body); } catch (err) { // jangan biarkan error di trigger mengganggu proses edit sheet biasa } } /** Bikin sheet "Ringkasan Utilisasi" yang otomatis menghitung % pemakaian tiap menu * dari data di Log_Penggunaan (pakai formula, jadi selalu update sendiri). */ function setupRingkasanUtilisasi(ss) { const name = 'Ringkasan Utilisasi'; let sh = ss.getSheetByName(name); if (!sh) sh = ss.insertSheet(name); sh.clear(); sh.getRange('A1').setValue('Ringkasan Utilisasi Portal GASPOL').setFontWeight('bold').setFontSize(14).setFontColor('#C81E2C'); sh.getRange('A3:C3').setValues([['Halaman/Fitur', 'Jumlah Akses', '% Utilisasi']]); sh.getRange('A3:C3').setFontWeight('bold').setBackground('#C81E2C').setFontColor('#FFFFFF'); const pages = ['calc', 'potongan', 'glossary', 'chatbot']; const labels = ['Gross-Up', 'Pemotongan', 'Glosarium', 'Tanya Pajak']; for (let i = 0; i < pages.length; i++) { const row = 4 + i; sh.getRange(row, 1).setValue(labels[i]); sh.getRange(row, 2).setFormula(`=COUNTIF('${LOG_SHEET_NAME}'!B:B,"${pages[i]}")`); sh.getRange(row, 3).setFormula(`=IFERROR(B${row}/SUM($B$4:$B$7),0)`); sh.getRange(row, 3).setNumberFormat('0.0%'); } sh.getRange('A9').setValue('Total Akses Tercatat').setFontWeight('bold'); sh.getRange('B9').setFormula('=SUM(B4:B7)').setFontWeight('bold'); sh.autoResizeColumns(1, 3); } /** Ubah nomor HP Indonesia ke format internasional 62xxxxxxxxxx (tanpa +) */ function normalizePhone(raw) { let digits = (raw || '').toString().replace(/[^0-9]/g, ''); if (digits.startsWith('0')) { digits = '62' + digits.substring(1); } else if (digits.startsWith('8')) { digits = '62' + digits; } else if (!digits.startsWith('62')) { digits = '62' + digits; } return digits; } function doPost(e) { try { const body = JSON.parse(e.postData.contents); if (body.action === 'log_usage') return handleLogUsage(body); if (body.action === 'check_status') return jsonResponse({ online: getCsOnline() }); if (body.action === 'check_answer') return handleCheckAnswer(body); // default: pertanyaan diteruskan ke CS return handleQuestion(body); } catch (err) { return jsonResponse({ success: false, error: err.toString() }); } } function handleLogUsage(body) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(LOG_SHEET_NAME); if (!sheet) { setupSheet(); sheet = ss.getSheetByName(LOG_SHEET_NAME); } sheet.appendRow([new Date(), (body.page || '').toString(), (body.detail || '').toString()]); return jsonResponse({ success: true }); } /** Baca status CS online/offline dari sheet "Status_CS" (dicentang manual oleh Tim Tax). */ function getCsOnline() { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sh = ss.getSheetByName(STATUS_SHEET_NAME); if (!sh) return false; return sh.getRange('B1').getValue() === true; } catch (err) { return false; } } /** Dipanggil berkala (polling) oleh chatbot selama CS online, untuk cek apakah * baris pertanyaan tertentu sudah dijawab (kolom Jawaban terisi). */ function handleCheckAnswer(body) { const row = parseInt(body.row, 10); if (!row) return jsonResponse({ answered: false }); const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet || row > sheet.getLastRow()) return jsonResponse({ answered: false }); const jawaban = sheet.getRange(row, COL_JAWABAN).getValue().toString().trim(); return jsonResponse({ answered: !!jawaban, jawaban: jawaban }); } function handleQuestion(body) { const nama = (body.nama || '').toString().trim(); const hpRaw = (body.hp || '').toString().trim(); const email = (body.email || '').toString().trim(); const pertanyaan = (body.pertanyaan || '').toString().trim(); const sumber = (body.sumber || 'GASPOL Chatbot').toString().trim(); if (!nama || !hpRaw || !email || !pertanyaan) { return jsonResponse({ success: false, error: 'Data tidak lengkap (nama/hp/email/pertanyaan wajib diisi).' }); } const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { setupSheet(); sheet = ss.getSheetByName(SHEET_NAME); } const phone62 = normalizePhone(hpRaw); const waFormula = '=HYPERLINK("https://wa.me/' + phone62 + '?text=' + encodeURIComponent('Halo ' + nama + ', terkait pertanyaan Anda di GASPOL: "' + pertanyaan + '"') + '","' + hpRaw + '")'; const emailFormula = '=HYPERLINK("mailto:' + email + '?subject=' + encodeURIComponent('Jawaban pertanyaan GASPOL Anda') + '&body=' + encodeURIComponent('Halo ' + nama + ',\n\nTerkait pertanyaan Anda: "' + pertanyaan + '"\n\n') + '","' + email + '")'; const row = [ new Date(), nama, hpRaw, waFormula, email, emailFormula, pertanyaan, sumber, 'Baru', '', '', '' ]; sheet.appendRow(row); const lastRow = sheet.getLastRow(); sheet.getRange(lastRow, 4).setFormula(waFormula); // WA Link sheet.getRange(lastRow, 6).setFormula(emailFormula); // Email Link notifyAdmin(nama, hpRaw, email, pertanyaan, sumber); return jsonResponse({ success: true, rowId: lastRow, online: getCsOnline() }); } /** Kirim email notifikasi ke Tim Tax setiap ada pertanyaan baru masuk dari chatbot. */ function notifyAdmin(nama, hp, email, pertanyaan, sumber) { if (!ADMIN_EMAIL || ADMIN_EMAIL.indexOf('isi-email-anda') !== -1) return; // belum dikonfigurasi, lewati try { const subject = '[GASPOL] Pertanyaan Baru dari ' + nama; const body = 'Ada pertanyaan baru masuk dari chatbot Tanya Pajak GASPOL:\n\n' + 'Nama : ' + nama + '\n' + 'No. WhatsApp: ' + hp + '\n' + 'Email : ' + (email || '-') + '\n' + 'Sumber : ' + sumber + '\n\n' + 'Pertanyaan:\n' + pertanyaan + '\n\n' + '---\n' + 'Balas lewat WhatsApp/Email penanya, lalu update kolom Status & Jawaban di sheet "Pertanyaan_CS".\n' + 'Mohon dijawab dalam 1×24 jam.'; MailApp.sendEmail(ADMIN_EMAIL, subject, body); } catch (err) { // gagal kirim email tidak boleh menggagalkan penyimpanan pertanyaan } } function doGet(e) { return jsonResponse({ status: 'ok', message: 'GASPOL Tanya Pajak backend aktif.' }); } /** Jalankan fungsi ini SECARA MANUAL (pilih dari dropdown, klik Run) untuk tes cepat * apakah kirim email sudah berfungsi. Beda dengan notifyAdmin(), fungsi ini SENGAJA * tidak dibungkus try/catch — kalau ada masalah (mis. izin belum di-Allow), errornya * akan langsung kelihatan di Execution log, jadi lebih mudah didiagnosis. */ function testKirimEmail() { if (!ADMIN_EMAIL || ADMIN_EMAIL.indexOf('isi-email-anda') !== -1) { throw new Error('ADMIN_EMAIL belum diisi! Isi dulu di baris atas file Code.gs, lalu Deploy > Manage deployments > New version > Deploy.'); } MailApp.sendEmail( ADMIN_EMAIL, '[GASPOL] Tes Email Berhasil', 'Ini email percobaan dari Apps Script GASPOL.\n\n' + 'Kalau Anda menerima email ini, berarti fitur notifikasi pertanyaan baru & auto-email jawaban sudah siap dipakai.' ); Logger.log('Email tes berhasil dikirim ke ' + ADMIN_EMAIL); } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
